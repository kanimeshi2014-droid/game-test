const map = new mapboxgl.Map({ container: 'map', style: 'mapbox://styles/mapbox/streets-v12', center: [139.767125, 35.681236], // 東京駅の座標（あとでお好みの場所へ変えられます） zoom: 16, pitch: 60 }); // 💡 「style.load」ではなく「load」というタイミングで動かすことで、白いバグを完全に防ぎます map.on('load', () => { // ① 地形（3Dの地面のボコボコ）を設定する map.addSource('mapbox-dem', { 'type': 'raster-dem', 'url': 'mapbox://mapbox.mapbox-terrain-dem-v1', 'tileSize': 512, 'maxzoom': 14 }); map.setTerrain({ 'source': 'mapbox-dem', 'exaggeration': 1.5 }); // ② 地図の文字を日本語にする処理 map.getStyle().layers.forEach(layer => { if (layer.layout && layer.layout['text-field']) { map.setLayoutProperty(layer.id, 'text-field', [ 'coalesce', ['get', 'name_ja'], ['get', 'name'] ]); } }); // ③ 建物（ビル）を3Dでニョキニョキ立ち上げる処理 map.addLayer({ 'id': '3d-buildings', 'source': 'composite', 'source-layer': 'building', 'filter': ['==', 'extrude', 'true'], 'type': 'fill-extrusion', 'minzoom': 15, 'paint': { 'fill-extrusion-color': '#aaa', // ビルの色 'fill-extrusion-height': [ 'interpolate', ['linear'], ['zoom'], 15, 0, 15.05, ['get', 'height'] ], 'fill-extrusion-base': ['get', 'min_height'], 'fill-extrusion-opacity': 0.6 } }); });
